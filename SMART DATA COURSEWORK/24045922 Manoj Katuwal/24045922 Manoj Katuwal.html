def show_stats(column, name):
    print(f"\nStatistics for {name}:")
    print("Sum:", column.sum())
    print("Mean:", column.mean())
    print("Standard Deviation:", column.std())
    print("Skewness:", column.skew())
    print("Kurtosis:", column.kurt())

# Hypothesis test 

from scipy.stats import ttest_ind

# Separate the two groups
normal = new_cleaned_df[new_cleaned_df['Label'] == 'Normal Traffic']['Flow Duration']
attack = new_cleaned_df[new_cleaned_df['Label'] == 'Attack']['Flow Duration']

# Perform independent t-test
t_stat, p_value = ttest_ind(normal, attack, equal_var=False)

print("T-Statistic:", t_stat)
print("P-Value:", p_value)

# Decision
if p_value < 0.05:
    print("Result: Reject the null hypothesis (significant difference exists).")
else:
    print("Result: Fail to reject the null hypothesis (no significant difference).")
